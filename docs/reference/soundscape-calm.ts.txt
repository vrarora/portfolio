// Calm pad version of src/components/story/audio/soundscape.ts, kept for comparison.
// Its layer timings match the earlier story beats; retune setProgress and moodAt to BEATS before using it.
import { bump, lerp, range } from "../engine/math";

type LayerName = "street" | "night" | "rain" | "dawn" | "music";

const TICK_MS = 250;
const CHORD_SECONDS = 11;
const CHORD_FADE = 5;

/** Chord voicings in MIDI, one pair per mood. The pad alternates within a mood. */
const MOODS = {
  street: [
    [50, 57, 61, 64, 66], // Dmaj9
    [43, 50, 54, 57, 59], // Gmaj9
  ],
  night: [
    [47, 54, 57, 62, 64], // Bm11
    [43, 50, 54, 59, 62], // Gmaj7
  ],
  dawn: [
    [50, 57, 64, 66, 69], // D add9, open
    [45, 52, 59, 61, 64], // A add9
  ],
} as const;

type Mood = keyof typeof MOODS;

const midiToHz = (m: number) => 440 * Math.pow(2, (m - 69) / 12);

const moodAt = (p: number): Mood => (p < 0.44 ? "street" : p < 0.86 ? "night" : "dawn");

type Voice = { oscs: OscillatorNode[]; gain: GainNode };

/**
 * Everything is synthesised; nothing is downloaded.
 * A slow pad follows the scene's mood, a rare soft note sits on top,
 * and each scene has a quiet, steady bed of its own.
 */
export class Soundscape {
  private readonly ctx: AudioContext;
  private readonly master: GainNode;
  private readonly reverb: ConvolverNode;
  private readonly tone: BiquadFilterNode;
  private readonly layers: Record<LayerName, GainNode>;
  private readonly noise: AudioBuffer;
  private timer: number | null = null;
  private voices: Voice[] = [];
  private mood: Mood = "street";
  private chordIndex = 0;
  private nextChord = 0;
  private lastChord = -Infinity;
  private nextNote = 0;
  private nextBird = 0;
  private progress = 0;

  constructor() {
    this.ctx = new AudioContext();
    this.master = this.ctx.createGain();
    this.master.gain.value = 0;
    this.master.connect(this.ctx.destination);

    this.reverb = this.ctx.createConvolver();
    this.reverb.buffer = this.impulse(5);
    const wet = this.ctx.createGain();
    wet.gain.value = 0.6;
    this.reverb.connect(wet).connect(this.master);

    this.noise = this.noiseBuffer();
    const make = () => {
      const g = this.ctx.createGain();
      g.gain.value = 0;
      g.connect(this.master);
      return g;
    };
    this.layers = { street: make(), night: make(), rain: make(), dawn: make(), music: make() };

    // One filter over the pad: darker at night, open at dawn.
    this.tone = this.ctx.createBiquadFilter();
    this.tone.type = "lowpass";
    this.tone.frequency.value = 1200;
    this.tone.Q.value = 0.3;
    this.tone.connect(this.layers.music);
    this.tone.connect(this.reverb);
    this.layers.dawn.connect(this.reverb);

    this.buildStreet();
    this.buildNight();
    this.buildRain();
    this.buildBreeze();
  }

  async start() {
    await this.ctx.resume();
    const now = this.ctx.currentTime;
    this.master.gain.cancelScheduledValues(now);
    this.master.gain.setTargetAtTime(0.7, now, 1.2);
    this.mood = moodAt(this.progress);
    this.nextChord = now;
    this.nextNote = now + 6;
    this.nextBird = now + 3;
    this.setProgress(this.progress, true);
    if (this.timer === null) this.timer = window.setInterval(() => this.schedule(), TICK_MS);
  }

  async stop() {
    const now = this.ctx.currentTime;
    this.master.gain.cancelScheduledValues(now);
    this.master.gain.setTargetAtTime(0, now, 0.4);
    if (this.timer !== null) {
      window.clearInterval(this.timer);
      this.timer = null;
    }
    await new Promise((resolve) => window.setTimeout(resolve, 1500));
    if (this.timer === null) {
      this.releaseVoices(0.1);
      await this.ctx.suspend();
    }
  }

  dispose() {
    if (this.timer !== null) window.clearInterval(this.timer);
    this.timer = null;
    void this.ctx.close();
  }

  setProgress(p: number, immediate = false) {
    this.progress = p;
    const targets: Record<LayerName, number> = {
      street: 1 - range(p, 0.36, 0.46),
      night: range(p, 0.44, 0.52) * (1 - range(p, 0.84, 0.94)),
      rain: bump(p, 0.47, 0.5, 0.55, 0.58),
      dawn: range(p, 0.86, 0.97),
      music: 1,
    };
    const now = this.ctx.currentTime;
    const tau = immediate ? 0.05 : 0.8;
    (Object.keys(targets) as LayerName[]).forEach((name) => {
      this.layers[name].gain.setTargetAtTime(targets[name], now, tau);
    });
    const night = bump(p, 0.44, 0.52, 0.84, 0.92);
    const dawn = range(p, 0.86, 1);
    this.tone.frequency.setTargetAtTime(lerp(lerp(1300, 700, night), 2600, dawn), now, tau * 2);

    // A change of scene brings in the next mood straight away, crossfaded.
    const mood = moodAt(p);
    if (mood !== this.mood) {
      this.mood = mood;
      this.chordIndex = 0;
      // Scrolling back and forth across a boundary should not restart the pad every frame.
      this.nextChord = Math.max(now, this.lastChord + 3);
    }
  }

  /* ---------- Scheduling ---------- */

  private schedule() {
    const now = this.ctx.currentTime;
    if (now >= this.nextChord) {
      this.playChord(MOODS[this.mood][this.chordIndex % 2], now);
      this.chordIndex += 1;
      this.lastChord = now;
      this.nextChord = now + CHORD_SECONDS;
    }
    if (now >= this.nextNote) {
      const chord = MOODS[this.mood][(this.chordIndex + 1) % 2];
      this.softNote(midiToHz(chord[3 + Math.floor(Math.random() * 2)] + 12), now + 0.05);
      this.nextNote = now + 7 + Math.random() * 6;
    }
    if (now >= this.nextBird) {
      if (this.progress > 0.88) this.bird(now + 0.1);
      this.nextBird = now + 4 + Math.random() * 5;
    }
  }

  private playChord(notes: readonly number[], t: number) {
    this.releaseVoices(CHORD_FADE);
    this.voices = notes.map((m, i) => {
      const gain = this.ctx.createGain();
      gain.gain.setValueAtTime(0, t);
      gain.gain.linearRampToValueAtTime(i === 0 ? 0.06 : 0.035, t + CHORD_FADE);
      gain.connect(this.tone);
      const freq = midiToHz(m);
      const oscs = [-4, 4].map((cents, k) => {
        const osc = this.ctx.createOscillator();
        osc.type = k === 0 ? "sine" : "triangle";
        osc.frequency.value = freq;
        osc.detune.value = cents;
        osc.connect(gain);
        osc.start(t);
        return osc;
      });
      return { oscs, gain };
    });
  }

  private releaseVoices(seconds: number) {
    const now = this.ctx.currentTime;
    for (const v of this.voices) {
      v.gain.gain.cancelScheduledValues(now);
      v.gain.gain.setValueAtTime(v.gain.gain.value, now);
      v.gain.gain.linearRampToValueAtTime(0, now + seconds);
      v.oscs.forEach((o) => o.stop(now + seconds + 0.1));
    }
    this.voices = [];
  }

  /** One quiet, rounded note, like a felt piano heard from the next room. */
  private softNote(freq: number, t: number) {
    const out = this.ctx.createGain();
    out.gain.setValueAtTime(0, t);
    out.gain.linearRampToValueAtTime(0.045, t + 0.03);
    out.gain.exponentialRampToValueAtTime(0.0001, t + 6);
    out.connect(this.tone);
    [1, 2].forEach((ratio, i) => {
      const osc = this.ctx.createOscillator();
      osc.frequency.value = freq * ratio;
      const g = this.ctx.createGain();
      g.gain.value = i === 0 ? 1 : 0.12;
      osc.connect(g).connect(out);
      osc.start(t);
      osc.stop(t + 6.2);
    });
  }

  private bird(t: number) {
    const pan = this.ctx.createStereoPanner();
    pan.pan.value = Math.random() * 1.2 - 0.6;
    pan.connect(this.layers.dawn);
    const base = 2600 + Math.random() * 900;
    const notes = 2 + Math.floor(Math.random() * 2);
    for (let i = 0; i < notes; i += 1) {
      const s = t + i * 0.16;
      const osc = this.ctx.createOscillator();
      const g = this.ctx.createGain();
      osc.frequency.setValueAtTime(base, s);
      osc.frequency.exponentialRampToValueAtTime(base * 1.25, s + 0.1);
      g.gain.setValueAtTime(0, s);
      g.gain.linearRampToValueAtTime(0.012, s + 0.03);
      g.gain.exponentialRampToValueAtTime(0.0001, s + 0.14);
      osc.connect(g).connect(pan);
      osc.start(s);
      osc.stop(s + 0.16);
    }
  }

  /* ---------- Beds ---------- */

  private loop(): AudioBufferSourceNode {
    const src = this.ctx.createBufferSource();
    src.buffer = this.noise;
    src.loop = true;
    src.start(0, Math.random() * 3);
    return src;
  }

  /** Slowly breathing gain, so a bed never sits perfectly still. */
  private breathe(target: AudioParam, rate: number, depth: number) {
    const lfo = this.ctx.createOscillator();
    lfo.frequency.value = rate;
    const amount = this.ctx.createGain();
    amount.gain.value = depth;
    lfo.connect(amount).connect(target);
    lfo.start();
  }

  /** A far-off market: a warm, low hum of people, heard through walls. */
  private buildStreet() {
    const filter = this.ctx.createBiquadFilter();
    filter.type = "bandpass";
    filter.frequency.value = 380;
    filter.Q.value = 0.8;
    const gain = this.ctx.createGain();
    gain.gain.value = 0.05;
    this.breathe(gain.gain, 0.09, 0.015);
    this.loop().connect(filter).connect(gain).connect(this.layers.street);
  }

  /** Night air: a soft wind and a faint, steady shimmer of insects. */
  private buildNight() {
    const wind = this.ctx.createBiquadFilter();
    wind.type = "lowpass";
    wind.frequency.value = 500;
    const windGain = this.ctx.createGain();
    windGain.gain.value = 0.06;
    this.breathe(windGain.gain, 0.05, 0.03);
    this.loop().connect(wind).connect(windGain).connect(this.layers.night);

    const shimmer = this.ctx.createBiquadFilter();
    shimmer.type = "bandpass";
    shimmer.frequency.value = 4600;
    shimmer.Q.value = 12;
    const shimmerGain = this.ctx.createGain();
    shimmerGain.gain.value = 0.012;
    this.breathe(shimmerGain.gain, 0.6, 0.006);
    this.loop().connect(shimmer).connect(shimmerGain).connect(this.layers.night);
  }

  private buildRain() {
    const lp = this.ctx.createBiquadFilter();
    lp.type = "lowpass";
    lp.frequency.value = 3000;
    const hp = this.ctx.createBiquadFilter();
    hp.type = "highpass";
    hp.frequency.value = 500;
    const gain = this.ctx.createGain();
    gain.gain.value = 0.07;
    this.loop().connect(hp).connect(lp).connect(gain).connect(this.layers.rain);
  }

  private buildBreeze() {
    const filter = this.ctx.createBiquadFilter();
    filter.type = "lowpass";
    filter.frequency.value = 700;
    const gain = this.ctx.createGain();
    gain.gain.value = 0.05;
    this.breathe(gain.gain, 0.07, 0.025);
    this.loop().connect(filter).connect(gain).connect(this.layers.dawn);
  }

  /* ---------- Buffers ---------- */

  private noiseBuffer() {
    const length = this.ctx.sampleRate * 6;
    const buffer = this.ctx.createBuffer(1, length, this.ctx.sampleRate);
    const data = buffer.getChannelData(0);
    // Brown-ish noise: softer and rounder than white.
    let last = 0;
    for (let i = 0; i < length; i += 1) {
      last = (last + 0.02 * (Math.random() * 2 - 1)) / 1.02;
      data[i] = last * 3.5;
    }
    return buffer;
  }

  private impulse(seconds: number) {
    const rate = this.ctx.sampleRate;
    const length = Math.floor(rate * seconds);
    const buffer = this.ctx.createBuffer(2, length, rate);
    for (let c = 0; c < 2; c += 1) {
      const data = buffer.getChannelData(c);
      for (let i = 0; i < length; i += 1) data[i] = (Math.random() * 2 - 1) * Math.pow(1 - i / length, 3);
    }
    return buffer;
  }
}
